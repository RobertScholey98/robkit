```ts title="addresses.ts"
import { installAt } from "@robkit/input/addresses";

  "core",
);

```

## React

```tsx
import { useState } from "react";

export function Counter() {
  const [count, setCount] = useState(0);
  return (
    <button type="button" onClick={() => setCount(count + 1)}>
      Clicked {count} times
    </button>
  );
}
```
